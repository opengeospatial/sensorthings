== JsonArray Format Extension

The JsonArray Format Extension describes a format that allows a more compact representation of data.

TODO: It also describes an interface that allows for more efficient upload of observations.

=== Requesting Data



.An example request using the JsonArray format.
[source,text]
----
/Datastreams(5)/Observations?
    $select=id,resultTime,phenomenonTime,result&
    $count=true&
    $format=jsonarray
----

.An example response, matching the above request, using the JsonArray format.
[source,json]
----
{
    "@context": "...",
    "structure": [
        "id",
        "resultTime",
        {
            "attribute": "phenomenonTime",   /* phenomenonTime is complex */
            "structure": [
                "start",
                "end"
            ]
        },
        "result"
    ],
    "@count": 4200,
    "value": [
        [1, "2026-01-01T00:00:00Z", ["2026-01-01T00:00:00Z", "2026-01-01T01:00:00Z"], 42],
        [2, "2026-01-01T01:00:00Z", ["2026-01-01T01:00:00Z", "2026-01-01T02:00:00Z"], 42],
        [3, "2026-01-01T02:00:00Z", ["2026-01-01T02:00:00Z", "2026-01-01T03:00:00Z"], 42],
        [4, "2026-01-01T03:00:00Z", ["2026-01-01T03:00:00Z", "2026-01-01T04:00:00Z"], 42],
        ...
    ],
    "@nextLink": "<nextLink for Observations>"
}
----


.An example request using the JsonArray format.
[source,text]
----
/Datastreams?
    $select=id,name,phenomenonTime/start,phenomenonTime/end&
    $count=true&
    $expand=
        ProximateFeatureOfIterest($select=id,name,properties),
        ObservedProperties($select=id,name;$count=false),
        Observations($select=id,phenomenonTime,result;$count=true)&
    $format=jsonarray
----

.An example response, matching the above request, using the JsonArray format.
[source,json]
----
{
    "@context": "...",
    "structure": [
        "id",
        "name",
        "phenomenonTime/start",
        "phenomenonTime/end",
        {                               /* Single entity ProximateFeatureOfIterest, properties is an open type */
            "attribute": "ProximateFeatureOfIterest",
            "structure": [
                "id",
                "name",
                "properties"
            ]
        },
        "ObservedProperties@count",     /* Count for EntitySet, if requested */
        {                               /* contents of EntitySet ObservedProperties, array of arrays */
            "attribute": "ObservedProperties",
            "structure": [
                "id",
                "name"
            ]
        },
        "ObservedProperties@nextLink",  /* NextLink for EntitySet, may be null */
        "Observations@count",           /* Count for EntitySet */
        {
            "attribute": "Observations",
            "structure": [              /* contents of EntitySet Observations */
                "id",
                {
                    "attribute": "phenomenonTime",   /* phenomenonTime is complex */
                    "structure": [
                        "start",
                        "end"
                    ]
                },
                "result"
            ]
        }
        "Observations@nextLink",        /* NextLink for EntitySet */
    ],
    "@count": 42,
    "value": [
        [
            1,
            "Datastream 1",
            "2026-01-01T00:00:00Z",     /* Because phenomenonTime/start is explicitly selected, it is extracted from the complex attribute */
            "2028-01-01T00:00:00Z",     /* Because phenomenonTime/end is explicitly selected, it is extracted from the complex attribute */
            [                           /* ProximateFeatureOfInterest */
                1,
                "Sampling point 1",
                { "property1": "value 1"}
            ],
            [                           /* ObservedProperties */
                [
                    1,
                    "Temperature"
                ]
            ],
            null,                       /* No nextLink for ObservedProperties */
            4200,                       /* Observations count */
            [
                [1, ["2026-01-01T00:00:00Z", "2026-01-01T01:00:00Z"], 42],
                [2, ["2026-01-01T01:00:00Z", "2026-01-01T02:00:00Z"], 42],
                [3, ["2026-01-01T02:00:00Z", "2026-01-01T03:00:00Z"], 42],
                [4, ["2026-01-01T03:00:00Z", "2026-01-01T04:00:00Z"], 42]
            ],
            "<nextLink for Observations>"
        ],
        [
            2,
            "Datastream 2",
            "2026-01-01T00:00:00Z",     /* Because phenomenonTime/start is explicitly selected, it is extracted from the complex attribute */
            "2028-01-01T00:00:00Z",     /* Because phenomenonTime/end is explicitly selected, it is extracted from the complex attribute */
            [                           /* ProximateFeatureOfInterest */
                2,
                "Sampling point 2",
                { "property1": "value 2"}
            ],
            [                           /* ObservedProperties */
                [
                    3,
                    "AirHumidity"
                ]
            ],
            null,                       /* No nextLink for ObservedProperties */
            4200,                       /* Observations count */
            [
                [1, ["2026-01-01T00:00:00Z", "2026-01-01T01:00:00Z"], 42],
                [2, ["2026-01-01T01:00:00Z", "2026-01-01T02:00:00Z"], 42],
                [3, ["2026-01-01T02:00:00Z", "2026-01-01T03:00:00Z"], 42],
                [4, ["2026-01-01T03:00:00Z", "2026-01-01T04:00:00Z"], 42]
            ],
            "<nextLink for Observations>"
        ],
    ],
    "@nextLink": "<nextLink for Datastreams>"
}
----
